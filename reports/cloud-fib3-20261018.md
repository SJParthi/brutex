`src/main.rs`:

```rust
//! FIB-3's real-data check (read-only): Zerodha one-minute bars of three
//! stock-months, folded per stock-day into every width 1..=N_max (17:15 IST
//! item 3), run through the base detector. Prints, per stock-month and width:
//! bases per kind and direction per day, every refusal counter, the stamps,
//! ten seeded random snapshots with every measurement, and the real-data
//! look-ahead check (every snapshot at candle j re-derived from the day cut
//! after j).
//!
//! Usage: fib3-real <store root> [<widths: e.g. 1,2,3,5,15 or all>]
//!
//! The detector's learned inputs come from the stock's own past in the real
//! run (FIB-2, FIB-9). Here, with no own past, both directions run with no
//! generated rest lengths and the one touch level 0 bp (exact touches), which
//! is printed; the measurement counts are the point. Nothing is written but
//! standard output.

use std::path::Path;

use brutex_core::vendor::Vendor;
use indicators::fibflag::base::{CandleInput, DayInput, Detector, DirectionDay};
use indicators::fibflag::extremes::WithheldExtremes;
use indicators::fibflag::types::{BaseKind, DetectorParams, FibCandle, Refusal, TradeDir, ZeroWidthReading};
use indicators::{Candle, ist_day};
use pull::calendar::{DayKind, kind_of};

const MINUTE: i64 = 60_000_000;
const IST: i64 = 19_800_000_000;

/// The stock-months the specification names.
const MONTHS: [(&str, u16, u8); 3] = [("HCLTECH", 2025, 10), ("RELIANCE", 2025, 3), ("TECHM", 2026, 8)];

struct Seeded(u64);
impl Seeded {
    fn next(&mut self) -> u64 {
        self.0 = self.0.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = self.0;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^ (z >> 31)
    }
}

/// One day's minutes, by IST minute of day.
fn minute_of_day(ts: i64) -> i64 {
    (ts + IST).rem_euclid(86_400_000_000) / MINUTE
}

/// A bucket of the width's grid: its open instant and the minutes present.
struct Bucket {
    open_ts: i64,
    close_ts: i64,
    minutes: Vec<Candle>,
    expected: i64,
}

/// The grid of one day at width `n`, anchored at each session window's open.
fn buckets(day: i64, minutes: &[Candle], n: i64) -> Option<Vec<Bucket>> {
    let DayKind::Open(session) = kind_of(day) else { return None };
    let midnight = day * 86_400_000_000 - IST;
    let mut out = Vec::new();
    for window in session.windows.iter().take(usize::from(session.count)) {
        let (from, to) = (i64::from(window.from), i64::from(window.to));
        let mut start = from;
        while start <= to {
            let end = (start + n - 1).min(to);
            let present: Vec<Candle> = minutes
                .iter()
                .filter(|c| (start..=end).contains(&minute_of_day(c.ts_micros)))
                .copied()
                .collect();
            out.push(Bucket {
                open_ts: midnight + start * MINUTE,
                close_ts: midnight + (end + 1) * MINUTE,
                minutes: present,
                expected: end - start + 1,
            });
            start += n;
        }
    }
    Some(out)
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let Some(root) = args.get(1) else {
        eprintln!("usage: fib3-real <store root> [widths]");
        std::process::exit(2);
    };
    let top = vocab_top();
    let widths: Vec<i64> = match args.get(2).map(String::as_str) {
        None | Some("all") => (1..=top).collect(),
        Some(list) => list.split(',').map(|w| w.parse().expect("a width")).collect(),
    };
    println!("fib3-real: store {root}; widths {widths:?}; touch tolerance 0 bp; no generated rest lengths");
    for (symbol, year, month) in MONTHS {
        let loaded = match cli::stored::load(Path::new(root), Vendor::Zerodha, symbol, "1min", year, month) {
            Ok(loaded) => loaded,
            Err(why) => {
                println!("{symbol} {year}-{month:02}: REFUSED: {why}");
                continue;
            }
        };
        println!("{symbol} {year}-{month:02}: {} minutes", loaded.bars.len());
        let mut days: Vec<(i64, Vec<Candle>)> = Vec::new();
        for bar in &loaded.bars {
            let day = ist_day(bar.ts_micros);
            match days.last_mut() {
                Some((d, list)) if *d == day => list.push(*bar),
                _ => days.push((day, vec![*bar])),
            }
        }
        let tick = observed_tick(&loaded.bars);
        println!("  observed tick {tick} paisa (gcd of the month's price differences)");
        for &n in &widths {
            run_width(symbol, &days, n, tick);
        }
    }
}

fn vocab_top() -> i64 {
    i64::try_from(vocab::fib_cube::max_width_minutes()).expect("a small width")
}

/// The observed tick of a stock-month: the gcd of every price's distance from
/// the first (FIB-2's ObservedTick reads it as a running gcd per series).
fn observed_tick(bars: &[Candle]) -> i64 {
    fn gcd(a: i64, b: i64) -> i64 {
        if b == 0 { a.abs() } else { gcd(b, a % b) }
    }
    let Some(first) = bars.first().map(|c| c.open) else { return 0 };
    bars.iter()
        .flat_map(|c| [c.open, c.high, c.low, c.close])
        .fold(0, |g, p| gcd(g, p - first))
}

fn run_width(symbol: &str, days: &[(i64, Vec<Candle>)], n: i64, tick: i64) {
    let grids: Vec<(i64, Vec<Bucket>)> = days
        .iter()
        .filter_map(|(day, minutes)| buckets(*day, minutes, n).map(|b| (*day, b)))
        .collect();
    let capacity = grids.iter().map(|(_, b)| b.len()).max().unwrap_or(0);
    let params = DetectorParams {
        tf_minutes: u16::try_from(n).expect("a width"),
        generation_version: 0,
        ring_capacity: u16::try_from(capacity).expect("a session"),
        long_rest_lengths: 0,
        short_rest_lengths: 0,
        long_touch_levels: 1,
        short_touch_levels: 1,
        zero_width: ZeroWidthReading::Emit,
    };
    let mut detector = Detector::new(params).expect("a width in 1..=N_max");
    let mut per_kind = [[0_u64; 3]; 2];
    let mut rng = Seeded(0x00F1_B3D0 ^ n as u64);
    let mut samples = Vec::new();
    let mut lookahead_checked = 0_u64;
    for (day, grid) in &grids {
        let input = DayInput {
            ist_day: *day,
            session_candles: u16::try_from(grid.len()).expect("a session"),
            observed_tick: tick,
            long: DirectionDay { rest_lengths: &[], touch_bp: Some(&[0]) },
            short: DirectionDay { rest_lengths: &[], touch_bp: Some(&[0]) },
        };
        if detector.begin_day(&input).is_err() {
            continue;
        }
        let mut honest = Vec::new();
        let inputs: Vec<Option<CandleInput>> = grid.iter().enumerate().map(|(k, b)| candle_of(b, grid.get(k + 1), n)).collect();
        for (bucket, input) in grid.iter().zip(&inputs) {
            match input {
                Some(input) => {
                    let emitted = detector.fold(input).map(|e| e.snapshots().to_vec()).unwrap_or_default();
                    for s in &emitted {
                        let d = usize::from(s.direction == TradeDir::Short);
                        let k = match s.kind { BaseKind::Micro2 => 0, BaseKind::Micro3 => 1, BaseKind::Rest => 2 };
                        per_kind[d][k] += 1;
                        if rng.next() % 4_000 == 0 && samples.len() < 10 {
                            samples.push(*s);
                        }
                    }
                    honest.push(emitted);
                }
                None if !bucket.minutes.is_empty() => {
                    let update = WithheldExtremes {
                        open_ts_micros: bucket.open_ts,
                        close_ts_micros: bucket.close_ts,
                        high: bucket.minutes.iter().map(|c| c.high).max().unwrap_or(0),
                        low: bucket.minutes.iter().map(|c| c.low).min().unwrap_or(0),
                        suspect_print: false,
                    };
                    let _ = detector.fold_withheld(&update);
                }
                None => {}
            }
        }
        // The real-data look-ahead check: the day re-run, cut after a random
        // candle j with every later candle dropped, gives candle j's
        // snapshots bit for bit.
        let folded = honest.len();
        if folded > 0 {
            let cut = usize::try_from(rng.next() % folded as u64).unwrap_or(0);
            let mut again = Detector::new(params).expect("a width");
            again.begin_day(&input).expect("the same day");
            let mut seen = 0;
            for (bucket, input) in grid.iter().zip(&inputs) {
                match input {
                    Some(input) if seen <= cut => {
                        let got = again.fold(input).map(|e| e.snapshots().to_vec()).unwrap_or_default();
                        assert_eq!(got, honest[seen], "{symbol} day {day} width {n}: candle {seen}");
                        seen += 1;
                    }
                    None if seen <= cut && !bucket.minutes.is_empty() => {
                        let _ = again.fold_withheld(&WithheldExtremes {
                            open_ts_micros: bucket.open_ts,
                            close_ts_micros: bucket.close_ts,
                            high: bucket.minutes.iter().map(|c| c.high).max().unwrap_or(0),
                            low: bucket.minutes.iter().map(|c| c.low).min().unwrap_or(0),
                            suspect_print: false,
                        });
                    }
                    _ => {}
                }
            }
            lookahead_checked += 1;
        }
    }
    let days = grids.len().max(1) as u64;
    let r = detector.refusals();
    let c = detector.counts();
    println!(
        "  {symbol} width {n}: {} days; per day long micro2 {} micro3 {} rest {}, short micro2 {} micro3 {} rest {}",
        grids.len(),
        per_kind[0][0] / days, per_kind[0][1] / days, per_kind[0][2] / days,
        per_kind[1][0] / days, per_kind[1][1] / days, per_kind[1][2] / days
    );
    let refusals: Vec<String> = Refusal::ALL.iter().map(|x| format!("{} {}", x.name(), r.get(*x))).collect();
    println!("    refusals: {}", refusals.join(", "));
    println!(
        "    stamps: zero-width {}, ratio over zero {}, withheld extreme {}, suspect frame {}, withheld updates {}, rest resets {}, touch recounts {} ({} candles); look-ahead days checked {lookahead_checked}",
        c.zero_width_bases, c.ratio_over_zero, c.extreme_from_withheld_bucket, c.frame_from_suspect_print,
        c.withheld_updates, c.rest_resets, c.touch_recounts, c.touch_recount_candles
    );
    for s in &samples {
        println!("    sample: {s:?}");
    }
}

/// The detector's candle for a complete bucket; `None` for one with a minute
/// missing (withheld, its present minutes an extremes-only update).
fn candle_of(bucket: &Bucket, next: Option<&Bucket>, n: i64) -> Option<CandleInput> {
    if i64::try_from(bucket.minutes.len()).ok()? != bucket.expected || bucket.expected != n {
        return None;
    }
    let first = bucket.minutes.first()?;
    let last = bucket.minutes.last()?;
    Some(CandleInput {
        candle: FibCandle {
            candle: Candle {
                ts_micros: bucket.open_ts,
                open: first.open,
                high: bucket.minutes.iter().map(|c| c.high).max()?,
                low: bucket.minutes.iter().map(|c| c.low).min()?,
                close: last.close,
                volume: bucket.minutes.iter().map(|c| c.volume).sum(),
                open_interest: last.open_interest,
            },
            tf_minutes: u16::try_from(n).ok()?,
            complete: true,
            missing_minutes: 0,
            close_ts_micros: bucket.close_ts,
        },
        next_close_ts_micros: next.map_or(bucket.close_ts + n * MINUTE, |b| b.close_ts),
        suspect_print: false,
    })
}
```
