   `src/main.rs`:

   ```rust
   //! FIB-2 real-data day check, READ-ONLY (a scratch program outside the
   //! repository). Loads Zerodha one-minute bars through `cli::stored::load` for
   //! every symbol directory under `<store>/bars/zerodha/NSE/CASH` — TODAY'S
   //! SHARES, SURVIVORSHIP-BIASED, DEVELOPMENT ONLY — for one month (and the month
   //! before it, for the first day's previous close), folds the mover rank minute
   //! by minute, writes the WHOLE ranking at every Zerodha minute close to
   //! `<out>/rank-<yyyy>-<mm>.txt`, prints a per-day census, and prints one
   //! symbol's AbsMoveBp terciles per day, read at every minute, to show they
   //! change only at day closes. Writes nothing but `<out>`, which must not exist.
   //!
   //! Usage: fib2-daycheck <store-root> <new-out-dir> <year> <month> <symbol> <tercile-year> <first-month> <last-month>

   use std::collections::BTreeMap;
   use std::fmt::Write as _;
   use std::path::Path;

   use brutex_core::vendor::Vendor;
   use indicators::fibflag::hist::OwnPast;
   use runner::fib::rank::MoverRank;

   const DAY: i64 = 86_400_000_000;
   const IST: i64 = 19_800_000_000;
   const MINUTE: i64 = 60_000_000;

   fn ist_day(ts: i64) -> i64 {
       (ts + IST).div_euclid(DAY)
   }

   /// A symbol's bars by IST day: (minute close instant, close) in time order.
   type Days = BTreeMap<i64, Vec<(i64, i64)>>;

   fn load(root: &Path, symbol: &str, year: u16, month: u8) -> Days {
       let mut days = Days::new();
       match cli::stored::load(root, Vendor::Zerodha, symbol, "1min", year, month) {
           Ok(loaded) => {
               for bar in loaded.bars {
                   days.entry(ist_day(bar.ts_micros))
                       .or_default()
                       .push((bar.ts_micros + MINUTE, bar.close));
               }
           }
           Err(why) => eprintln!("{symbol} {year}-{month:02}: refused: {why}"),
       }
       days
   }

   fn previous_month(year: u16, month: u8) -> (u16, u8) {
       if month == 1 { (year - 1, 12) } else { (year, month - 1) }
   }

   fn main() {
       let args: Vec<String> = std::env::args().collect();
       let [_, store, out, year, month, symbol, t_year, first, last] = &args[..] else {
           eprintln!("usage: fib2-daycheck <store-root> <new-out-dir> <year> <month> <symbol> <tercile-year> <first-month> <last-month>");
           std::process::exit(2);
       };
       let root = Path::new(store);
       let out = Path::new(out);
       assert!(!out.exists(), "the out directory must be NEW");
       std::fs::create_dir_all(out).unwrap();
       let year: u16 = year.parse().unwrap();
       let month: u8 = month.parse().unwrap();

       // The pool: every symbol directory, sorted by name (development label; the
       // run's real tie rule is PIT-1's series-id order).
       let cash = root.join("bars/zerodha/NSE/CASH");
       let mut symbols: Vec<String> = std::fs::read_dir(&cash)
           .unwrap()
           .filter_map(|entry| entry.ok())
           .filter(|entry| entry.path().is_dir())
           .map(|entry| entry.file_name().to_string_lossy().into_owned())
           .collect();
       symbols.sort();
       println!(
           "pool: {} symbol directories under {} (today's shares, survivorship-biased, development only)",
           symbols.len(),
           cash.display()
       );
       let (py, pm) = previous_month(year, month);
       let mut series: Vec<Days> = Vec::with_capacity(symbols.len());
       for symbol in &symbols {
           let mut days = load(root, symbol, py, pm);
           days.extend(load(root, symbol, year, month));
           series.push(days);
       }
       let mut all_days: Vec<i64> = series.iter().flat_map(|d| d.keys().copied()).collect();
       all_days.sort_unstable();
       all_days.dedup();
       let mut rank = MoverRank::new(symbols.len()).unwrap();
       let mut text = String::new();
       println!("width {} for a pool of {}", rank.width().name(), symbols.len());
       for (at, day) in all_days.iter().enumerate().skip(1) {
           let (dy, dm) = civil(*day);
           if (dy, dm) != (year, month) {
               continue;
           }
           let previous_day = all_days[at - 1];
           // previous close = the previous tradable day's last Zerodha 1-minute close
           let previous: Vec<Option<i64>> = series
               .iter()
               .map(|days| days.get(&previous_day).and_then(|bars| bars.last()).map(|b| b.1))
               .collect();
           rank.begin_day(&previous).unwrap();
           let mut minutes: Vec<i64> = series
               .iter()
               .filter_map(|days| days.get(day))
               .flat_map(|bars| bars.iter().map(|b| b.0))
               .collect();
           minutes.sort_unstable();
           minutes.dedup();
           let mut cursor = vec![0_usize; series.len()];
           let (mut ranked_min, mut ranked_max, mut missing, mut stale) = (usize::MAX, 0, 0_u64, 0_u64);
           for minute in &minutes {
               let closes: Vec<Option<i64>> = series
                   .iter()
                   .zip(cursor.iter_mut())
                   .map(|(days, at)| {
                       let bars = days.get(day)?;
                       match bars.get(*at) {
                           Some((m, close)) if m == minute => {
                               *at += 1;
                               Some(*close)
                           }
                           _ => None,
                       }
                   })
                   .collect();
               let census = rank.minute(*minute, &closes).unwrap();
               ranked_min = ranked_min.min(usize::from(census.ranked));
               ranked_max = ranked_max.max(usize::from(census.ranked));
               missing += u64::from(census.missing);
               stale += u64::from(census.stale);
               let _ = write!(
                   text,
                   "{} pool {} ranked {} missing {} stale {} refused {} | G",
                   minute, census.pool, census.ranked, census.missing, census.stale, census.refused
               );
               for slot in rank.gainers() {
                   let _ = write!(text, " {}:{}", symbols[usize::from(*slot)], rank.move_bp(*slot).unwrap());
               }
               let _ = write!(text, " | L");
               for slot in rank.losers() {
                   let _ = write!(text, " {}:{}", symbols[usize::from(*slot)], rank.move_bp(*slot).unwrap());
               }
               text.push('\n');
           }
           println!(
               "{dy}-{dm:02}-{:02}: {} minutes; ranked per minute {}..={}; missing {missing} and stale {stale} member-minutes",
               day_of_month(*day),
               minutes.len(),
               if minutes.is_empty() { 0 } else { ranked_min },
               ranked_max
           );
       }
       std::fs::write(out.join(format!("rank-{year}-{month:02}.txt")), text).unwrap();

       // One symbol's AbsMoveBp terciles per day: read at every minute, they
       // change only at day closes.
       let t_year: u16 = t_year.parse().unwrap();
       let (first, last): (u8, u8) = (first.parse().unwrap(), last.parse().unwrap());
       let mut days = Days::new();
       let (py, pm) = previous_month(t_year, first);
       days.extend(load(root, symbol, py, pm));
       for m in first..=last {
           days.extend(load(root, symbol, t_year, m));
       }
       let mut past = OwnPast::empty();
       let mut previous_close: Option<i64> = None;
       for (day, bars) in &days {
           let mut seen = std::collections::BTreeSet::new();
           if let Some(previous) = previous_close {
               for (_, close) in bars {
                   let read = (past.quantile(1, 3).ok(), past.quantile(2, 3).ok());
                   seen.insert(read);
                   let bp = (i128::from(*close) - i128::from(previous)) * 10_000 / i128::from(previous);
                   past.observe(i64::try_from(bp.unsigned_abs()).unwrap()).unwrap();
               }
               past.commit_day().unwrap();
               let (dy, dm) = civil(*day);
               if (dy, dm) >= (t_year, first) {
                   println!(
                       "{symbol} {dy}-{dm:02}-{:02}: terciles read during the day {:?} (distinct readings: {}); after the close {:?}",
                       day_of_month(*day),
                       seen.iter().next(),
                       seen.len(),
                       (past.quantile(1, 3).ok(), past.quantile(2, 3).ok())
                   );
               }
           }
           previous_close = bars.last().map(|b| b.1);
       }
   }

   /// (year, month) of an IST day number (days since 1970-01-01).
   fn civil(day: i64) -> (u16, u8) {
       let (y, m, _) = ymd(day);
       (y, m)
   }

   fn day_of_month(day: i64) -> u8 {
       ymd(day).2
   }

   /// Howard Hinnant's civil-from-days.
   fn ymd(z: i64) -> (u16, u8, u8) {
       let z = z + 719_468;
       let era = z.div_euclid(146_097);
       let doe = z - era * 146_097;
       let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
       let y = yoe + era * 400;
       let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
       let mp = (5 * doy + 2) / 153;
       let d = doy - (153 * mp + 2) / 5 + 1;
       let m = if mp < 10 { mp + 3 } else { mp - 9 };
       let y = if m <= 2 { y + 1 } else { y };
       (y as u16, m as u8, d as u8)
   }
   ```
